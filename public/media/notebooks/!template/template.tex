\subsection{Theorems and Proofs}
\begin{theorem}
For any two real numbers \(a\) and \(b\), the following inequality holds:
\[
(a+b)^2 \geq 4ab.
\]
\end{theorem}

\begin{proof}
Expanding \( (a - b)^2 \geq 0 \), we get:
\[
a^2 - 2ab + b^2 \geq 0 \implies a^2 + b^2 \geq 2ab \implies (a + b)^2 \geq 4ab.
\]
\end{proof}

\subsection{Example}
\begin{example}
Consider the function \(f(x) = x^2\). For \(x = 2\), we have \(f(2) = 4\).
\end{example}
